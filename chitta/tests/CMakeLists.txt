cmake_minimum_required(VERSION 3.16)
project(chitta_ssl_tests LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
add_library(ssl_parser_under_test ../src/ssl_parser.cpp ../src/temporal.cpp)
target_include_directories(ssl_parser_under_test PUBLIC ../include)
# temporal.cpp normally gets this dependency from the daemon's includes.
target_compile_options(ssl_parser_under_test PRIVATE -include array)
add_executable(ssl_parser_test ssl_parser_test.cpp)
target_link_libraries(ssl_parser_test PRIVATE ssl_parser_under_test)
target_compile_options(ssl_parser_test PRIVATE -UNDEBUG)
target_compile_definitions(ssl_parser_test PRIVATE SSL_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
add_test(NAME ssl_parser_test COMMAND ssl_parser_test)
add_executable(parser_cli ../../benchmarks/distill/parser_cli.cpp)
target_link_libraries(parser_cli PRIVATE ssl_parser_under_test)
